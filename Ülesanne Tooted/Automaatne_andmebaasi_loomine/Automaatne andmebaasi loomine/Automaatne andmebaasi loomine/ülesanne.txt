🟦 4. Andmebaasi kontekst (ToodedContext.cs)
Loo fail ToodedContext.cs:

using Microsoft.EntityFrameworkCore;

public class ToodedContext : DbContext
{
    public DbSet<Toode> Toodetabel { get; set; }
    public DbSet<Kategooria> Kategooriatabel { get; set; }

    protected override void OnConfiguring(DbContextOptionsBuilder optionsBuilder)
    {
        optionsBuilder.UseSqlServer(
            @"Server=(localdb)\MSSQLLocalDB;Database=Tooded_DB;Trusted_Connection=True;");
    }

    protected override void OnModelCreating(ModelBuilder modelBuilder)
    {
        modelBuilder.Entity<Kategooria>().ToTable("Kategooriatabel");
        modelBuilder.Entity<Toode>().ToTable("Toodetabel");
    }
}

{
    dataGridViewTooted.DataSource = _db.Toodetabel
        .Include(t => t.Kategooria)
        .Select(t => new
        {
            t.Id,
            t.Toodenimetus,
            t.Kogus,
            t.Hind,
            Kategooria = t.Kategooria.Kategooria_nimetus,
            t.Pilt
        })
        .ToList();
}

private void btnLisa_Click(object sender, EventArgs e)
{
    var uus = new Toode
    {
        Toodenimetus = txtNimetus.Text,
        Kogus = int.Parse(txtKogus.Text),
        Hind = float.Parse(txtHind.Text),
        KategooriaId = (int)comboKategooria.SelectedValue,
        Pilt = Path.GetFileName(valitudPilt)
    };

    // kopeerime pildi projekti kausta
    if (!string.IsNullOrEmpty(valitudPilt))
    {
        File.Copy(valitudPilt, Path.Combine("Images", uus.Pilt), true);
    }

    _db.Toodetabel.Add(uus);
    _db.SaveChanges();
    LaeTooted();
}

private void dataGridViewTooted_CellClick(object sender, DataGridViewCellEventArgs e)
{
    if (e.RowIndex < 0) return;

    txtNimetus.Text = dataGridViewTooted.Rows[e.RowIndex].Cells["Toodenimetus"].Value.ToString();
    txtKogus.Text = dataGridViewTooted.Rows[e.RowIndex].Cells["Kogus"].Value.ToString();
    txtHind.Text = dataGridViewTooted.Rows[e.RowIndex].Cells["Hind"].Value.ToString();

    string pilt = dataGridViewTooted.Rows[e.RowIndex].Cells["Pilt"].Value?.ToString();

    if (!string.IsNullOrEmpty(pilt) && File.Exists(Path.Combine("Images", pilt)))
        picturePilt.Image = Image.FromFile(Path.Combine("Images", pilt));
}

⭐ 4. Lae kategooriad + loo TabPage + lisa pildid
private void LaeKategooriadJaPildid()
{
    tabPood.TabPages.Clear();

    var kategooriad = _db.Kategooriatabel.ToList();

    foreach (var kat in kategooriad)
    {
        // Loo uus tab selle kategooria jaoks
        var page = new TabPage(kat.Kategooria_nimetus);
        page.AutoScroll = true;

        // Too kategooria tooted andmebaasist
        var tooted = _db.Toodetabel
                        .Where(t => t.KategooriaId == kat.Id)
                        .ToList();

        int x = 10;
        int y = 10;
        int count = 0;

        foreach (var t in tooted)
        {
            // Loo paneel iga toote jaoks
            Panel p = new Panel();
            p.Width = 150;
            p.Height = 200;
            p.BorderStyle = BorderStyle.FixedSingle;

            // Pildikast
            PictureBox pb = new PictureBox();
            pb.Width = 140;
            pb.Height = 140;
            pb.SizeMode = PictureBoxSizeMode.Zoom;
            pb.Location = new Point(5, 5);

            // lae pilt failist, kui olemas
            var piltPath = Path.Combine("Images", t.Pilt ?? "puudub.png");
            if (File.Exists(piltPath))
                pb.Image = Image.FromFile(piltPath);

            // Nimetus
            Label lbl = new Label();
            lbl.Text = t.Toodenimetus;
            lbl.AutoSize = false;
            lbl.TextAlign = ContentAlignment.MiddleCenter;
            lbl.Location = new Point(5, 150);
            lbl.Width = 140;

            // Lisa elemendid paneelile
            p.Controls.Add(pb);
            p.Controls.Add(lbl);

            // Paiguta paneel TabPage-s
            p.Location = new Point(x, y);
            page.Controls.Add(p);

            // Järgmise positsioon
            x += 160;
            count++;

            // iga 4 pildi järel uus rida
            if (count % 4 == 0)
            {
                x = 10;
                y += 210;
            }
        }

        // Lisa TabPage TabControli sisse
        tabPood.TabPages.Add(page);
    }
}

🟦 6. PDF genereerimine (PdfSharpCore abil)

using PdfSharpCore.Drawing;
using PdfSharpCore.Pdf;

private void btnGenPDF_Click(object sender, EventArgs e)
{
    Directory.CreateDirectory("Arved");

    string failinimi = $"Arved/Arve_{DateTime.Now:yyyyMMdd_HHmmss}.pdf";

    PdfDocument document = new PdfDocument();
    PdfPage page = document.AddPage();
    XGraphics gfx = XGraphics.FromPdfPage(page);

    XFont font = new XFont("Verdana", 12);

    gfx.DrawString("Arve", new XFont("Verdana", 20), XBrushes.Black,
        new XRect(0, 20, page.Width, 40), XStringFormats.TopCenter);

    int y = 80;
    foreach (var item in ostukorv)
    {
        gfx.DrawString(item, font, XBrushes.Black,
            new XRect(20, y, page.Width - 40, 20), XStringFormats.TopLeft);
        y += 30;
    }

    document.Save(failinimi);

    MessageBox.Show($"PDF on loodud:\n{failinimi}");
}

using System.Net;
using System.Net.Mail;

private void btnSaadaEmail_Click(object sender, EventArgs e)
{
    string email = Microsoft.VisualBasic.Interaction.InputBox(
        "Sisesta e-mail", "Saaja", "nimi@example.com");

    if (string.IsNullOrEmpty(email)) return;

    string pdfPath = Directory.GetFiles("Arved")
        .OrderByDescending(f => f)
        .First();

    try
    {
        var mail = new MailMessage();
        mail.From = new MailAddress("sinu_gmail@gmail.com");
        mail.To.Add(email);
        mail.Subject = "Arve";
        mail.Body = "Arve on manuses.";
        mail.Attachments.Add(new Attachment(pdfPath));

        var smtp = new SmtpClient("smtp.gmail.com", 587)
        {
            Credentials = new NetworkCredential("sinu_gmail@gmail.com", "parool"),
            EnableSsl = true
        };

        smtp.Send(mail);

        MessageBox.Show("Arve saadetud!");
    }
    catch (Exception ex)
    {
        MessageBox.Show("Viga: " + ex.Message);

public class OstukorviRida

private void btnLisaKorvi_Click(object sender, EventArgs e)
{
    if (SelectedToode == null)
    {
        MessageBox.Show("Vali toode enne ostukorvi lisamist.");
        return;
    }

    string kogusStr = Microsoft.VisualBasic.Interaction.InputBox(
        "Sisesta kogus", "Kogus", "1");

    if (!int.TryParse(kogusStr, out int kogus) || kogus <= 0)
    {
        MessageBox.Show("Vale kogus.");
        return;
    }

    var rida = new OstukorviRida
    {
        Toode = SelectedToode,
        Kogus = kogus
    };

    ostukorv.Add(rida);
    listOstukorv.Items.Add(rida);

    ArvutaKoguSumma();
}

private void btnLogin_Click(object sender, EventArgs e)
{
    using var db = new ToodedContext();

    string email = txtEmail.Text;
    string parool = txtParool.Text;

    var kasutaja = db.Kasutajad.FirstOrDefault(k => k.Email == email);

    if (kasutaja == null)
    {
        lblVeateade.Text = "Vale email!";
        return;
    }

    if (!PasswordHelper.Verify(parool, kasutaja.ParoolHash))
    {
        lblVeateade.Text = "Vale parool!";
        return;
    }

    // roll salvestatakse globaalselt
    AktiivneKasutaja.Roll = kasutaja.Roll;

    // peale login
    FormMain main = new FormMain();
    main.Show();
    this.Hide();
}
